===== src/main/resources/templates/Post/list.html =====
<!DOCTYPE html>
<html lang="ko" xmlns:th="http://www.thymeleaf.org">
<head><meta charset="UTF-8"><meta name="viewport" content="width=device-width,initial-scale=1"><title>게시글 목록</title><link rel="stylesheet" th:href="@{/css/style.css}"></head>
<body><main><div class="eyebrow">SPRING BOARD</div>
<h1>게시글 목록</h1><div class="toolbar"><span>총 <strong th:text="${page.totalCount}">0</strong>개의 글</span><a class="button" th:href="@{/Post/insert}">글쓰기</a></div>
<table><thead><tr><th>번호</th><th>제목</th><th>작성자</th><th class="date">등록일</th></tr></thead><tbody>
<tr th:if="${#lists.isEmpty(posts)}"><td colspan="4">등록된 게시글이 없습니다.</td></tr>
<tr th:each="post : ${posts}"><td th:text="${post.postNo}"></td><td class="title"><a th:href="@{/Post/read(PostNo=${post.postNo})}" th:text="${post.title}"></a></td><td th:text="${post.writer}"></td><td class="date" th:text="${#temporals.format(post.createdAt, 'yyyy-MM-dd')}"></td></tr>
</tbody></table>
<nav aria-label="페이지 이동"><a th:if="${page.hasPrevious}" th:href="@{/Post/list(page=${page.page - 1})}">이전</a>
<th:block th:each="n : ${#numbers.sequence(page.startPage, page.endPage)}"><strong th:if="${n == page.page}" th:text="${n}" aria-current="page"></strong><a th:unless="${n == page.page}" th:href="@{/Post/list(page=${n})}" th:text="${n}"></a></th:block>
<a th:if="${page.hasNext}" th:href="@{/Post/list(page=${page.page + 1})}">다음</a></nav><a th:href="@{/}">홈으로</a>
</main></body></html>

<!DOCTYPE html>
<html lang="ko" xmlns:th="http://www.thymeleaf.org">
<head><meta charset="UTF-8"><meta name="viewport" content="width=device-width,initial-scale=1"><title>게시글 등록</title><link rel="stylesheet" th:href="@{/css/style.css}"></head>
